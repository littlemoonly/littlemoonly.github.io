{{- $u := urls.Parse .Destination -}} {{- $url := $u.String -}} {{- $imgResource := .Page.Scratch.Get "typoNilVariable" -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- if .PageInner.Resources -}} {{- $imgResource = .PageInner.Resources.Get $path -}} {{- else if (or .PageInner.Parent .PageInner.Parent.Resources) -}} {{- $imgResource = .PageInner.Parent.Resources.Get $path -}} {{- end -}} {{- $imgResource = or $imgResource (resources.Get $path) -}} {{- with $imgResource -}} {{- $url = .RelPermalink -}} {{- with $u.RawQuery -}} {{- $url = printf "%s?%s" $url . -}} {{- end -}} {{- with $u.Fragment -}} {{- $url = printf "%s#%s" $url . -}} {{- end -}} {{- end -}} {{- end -}} {{- $url = $url | safeURL -}} {{- $fileNameParts := split $url "#" -}} {{- $classes := "" -}} {{- range $idx := seq (sub (len $fileNameParts) 1) -}} {{- $tag := index $fileNameParts $idx -}} {{- $classes = printf "%s img-%s" $classes $tag -}} {{- end -}}
